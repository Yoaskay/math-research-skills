\begin{theorem}\label{th:existence}
Every contraction on a nonempty complete metric space has a fixed point.
\end{theorem}

\begin{proof}
Iterate the contraction from an arbitrary point, use the geometric estimate to obtain a Cauchy sequence, and pass to the limit.
\end{proof}

\begin{theorem}\label{th:stability}
If $T$ is a contraction on a complete metric space, then its fixed point is unique.
\end{theorem}

\begin{proof}
If $x$ and $y$ are fixed, then $d(x,y)=d(Tx,Ty)\leq c d(x,y)$ for some $c<1$. Hence $d(x,y)=0$.
\end{proof}
